# 背包

## 01背包
### 板板代码
```cpp
const int N = 1e5+2;
int n, c;
int w[N], v[N];

void go_01bag() {
    cin >> n >> c;
    for(int i=0; i<n; i++) {
        cin >> v[i] >> w[i];
    }
    // dp[j]：容量j背包能装的最大价值（背包不必装满，初始0）
    vector<int> dp(c+1, 0);
    for(int i=0; i<n; i++) {
        // 倒序遍历：保证 dp[j-v[i]] 是上一轮旧状态，当前物品只会选一次
        for(int j=c; j>=v[i]; j--) {
            dp[j] = max(dp[j], dp[j-v[i]] + w[i]);
        }
    }
    cout << dp[c] << '\n';
}
```
注意：上面代码是背包可以不满时的板子，若题目要求背包恰好装满，则需要改写dp初始化
```cpp
    vector<int> dp(c+1, INT_MIN);
    dp[0]=0;
```

## 多重背包
### 二进制拆分优化板板代码
```cpp
const int N = 1e5+2;
int n, c;            // n物品种类，c背包容量
int w[N], v[N], m[N];// w价值，v体积，m[i]物品i数量上限

void go() {
    cin >> n >> c;
    vector<int> nw, nv;// 拆分后的价值、体积数组
    int nn = 0; // 拆分后物品总个数（nv.size()等价，可选）
    for(int i=0; i<n; i++) {
        cin >> v[i] >> w[i] >> m[i];
    }

    for(int i=0; i<n; i++) {
        // 最多可取件数：物品数量上限 和 背包容纳上限取小
        int mi = m[i];
        // 如果题目有 v[i] 为正数 则可以 mi=min(c/v[i], m[i]);
        // 二进制拆分，转为01背包物品
        for(int j=1; j<=mi; j<<=1) {
            mi -= j;
            nn++;
            nv.push_back(j * v[i]);
            nw.push_back(j * w[i]);
        }
        if(mi) { // 剩余部分单独一组
            nn++;
            nv.push_back(mi * v[i]);
            nw.push_back(mi * w[i]);
        }
    }

    vector<int> dp(c+1, 0); // dp[j]:容量j最大价值，不必装满
    for(int i=0; i<nn; i++) {
        for(int j=c; j>=nv[i]; j--) { // 倒序，01背包
            dp[j] = max(dp[j], dp[j-nv[i]] + nw[i]);
        }
    }
    cout << dp[c] << '\n';
}
```

## 完全背包
### 板板代码
```cpp
const int N = 1e5+2;
int n, c;            // n：物品种类，c：背包总容量
int w[N], v[N];      // v[i]体积，w[i]价值，每种物品无限选取

void go() {
    cin >> n >> c;
    for(int i=0; i<n; i++) {
        cin >> v[i] >> w[i];
    }
    // dp[j]：背包容量j时，可获得的最大价值，背包不必装满，初始0
    vector<int> dp(c+1, 0);
    for(int i=0; i<n; i++) {
        // 正序遍历：复用本轮已经更新的dp[j-v[i]]，支持物品多次选取
        for(int j=v[i]; j<=c; j++) {
            dp[j] = max(dp[j], dp[j-v[i]]+w[i]);
        }
    }
    cout << dp[c] << '\n';
}
```
### 恰好装满板板代码
```cpp
const int N = 1e5+2;
int n, c;
int w[N], v[N];

void go_exact() {
    cin >> n >> c;
    for(int i=0; i<n; i++) {
        cin >> v[i] >> w[i];
    }
    // dp[j]：恰好装满容量j的最大价值
    vector<int> dp(c+1, INT_MIN);
    dp[0] = 0;
    for(int i=0; i<n; i++) {
        for(int j=v[i]; j<=c; j++) {
            dp[j] = max(dp[j], dp[j-v[i]]+w[i]);
        }
    }
    if(dp[c] < 0) cout << "no solution\n";
    else cout << dp[c] << '\n';
}
```
