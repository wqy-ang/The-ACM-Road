# 差分
## 一维差分板板代码
```cpp
#include <bits/stdc++.h>
using namespace std;
#define int long long

const int N = 1e5+5;
// 用差分，数组下标从 1 开始
vector<int> a(N);

void this_is_difference(int n) {
    // dif[i] : 本位与前一位的差，a[i] - a[i-1]
    vector<int> dif(n+2);
    for(int i=1; i<=n; i++) {
        dif[i] = a[i] - a[i-1];
    }

    // 区间 [l, r] 加val
    auto add = [&](int l, int r, int val){
        dif[l] += val;
        dif[r+1] -= val;
        // 注意是 r+1
    };

    // 还原数组，求前缀和
    int s = 0;
    for(int i=1; i<=n; i++) {
        // 先加 dif[i]，s 就表示第 i 位的数组值
        s += dif[i];
        a[i] = s;
    }
}
```
